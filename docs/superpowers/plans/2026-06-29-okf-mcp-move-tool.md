# okf-mcp `move` Tool Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `move` MCP tool that relocates an OKF concept to a new id, rewrites the markdown links so the bundle stays internally consistent, regenerates indexes, and records the change in `log.md`.

**Architecture:** One new pure primitive in `okf-core` (`rewrite_links`) recomputes a document's markdown links for a possibly-changed source directory and a target-id remap, reusing the exact resolution rules of `extract_links`. A new `operations.move_concept` in `okf-mcp` orchestrates: validate ids → relink the moved doc's own body → move the file → relink every inbound concept → regenerate indexes → append `log.md` entries. A thin `@mcp.tool() move` wrapper delegates to it. `okf-core` gains read-only graph logic only (no authoring), and all I/O orchestration stays in `okf-mcp`.

**Tech Stack:** Python 3.12, uv workspace, `okf-core` (PyYAML only), `okf-mcp` (official `mcp` SDK / FastMCP, `pydantic-settings`), pytest.

## Global Constraints

- Target Python `>=3.12`; uv workspace (`members = ["packages/*"]`). Run tests with `uv run pytest`.
- Concept IDs are bundle-relative paths without `.md`; reject final segments `index`/`log` (reserved files, RSV-01) and invalid segments — handled by the existing `operations._concept_path`.
- A move **preserves** the moved concept's frontmatter verbatim, including `timestamp` (a move is not a content edit). Inbound concepts whose links are rewritten also keep their frontmatter verbatim.
- Link rewriting reuses the resolution rules of `okf_core.extract_links`: the same `](target.md)` markdown form with an optional `#anchor`; relative targets resolve against the doc's directory, absolute (`/…`) against the bundle root; `scheme://` and out-of-bundle targets are left untouched.
- Only concept documents are rewritten. `index.md` is regenerated by `regenerate_indexes`; `log.md` is append-only history and is never rewritten.
- Log date headings are ISO `YYYY-MM-DD`, newest-first (reuse the existing `_append_log`).

```python
"""Rewrite a document's markdown links when it moves within a bundle."""
from __future__ import annotations

import os
import re
from pathlib import Path

from okf_core.paths import concept_id_to_path, parse_concept_id

# Mirrors bundle._LINK_RE but captures the target and anchor so they can be
# re-rendered. Keep the two patterns in sync.
_LINK_RE = re.compile(r"\]\((?P<target>[^)\s]+\.md)(?P<anchor>#[A-Za-z0-9_\-]*)?\)")


def _resolve_target_id(target: str, doc_dir: Path, bundle_root: Path) -> str | None:
    """Resolve a markdown link target to a bundle concept id, or ``None``.

    Returns ``None`` for external (``scheme://``) links and targets that escape
    the bundle root, matching ``extract_links``' skip rules.
    """
    if "://" in target:
        return None
    if target.startswith("/"):
        base, rel_target = bundle_root, target.lstrip("/")
    else:
        base, rel_target = doc_dir, target
    try:
        resolved = (base / rel_target).resolve().relative_to(bundle_root)
    except ValueError:
        return None
    rel = resolved.as_posix()
    if rel.endswith(".md"):
        rel = rel[:-3]
    return rel or None


def rewrite_links(
    body: str,
    doc_dir_old: Path,
    doc_dir_new: Path,
    bundle_root: Path,
    id_map: dict[str, str],
) -> str:
    """Recompute markdown links so they stay valid after a document moves.

    Resolves each ``](target.md)`` link in ``body`` (against ``doc_dir_old`` and
    ``bundle_root``, using the same rules as ``extract_links``), remaps the
    target concept id via ``id_map`` (old id -> new id), and re-renders the link
    relative to ``doc_dir_new`` -- preserving the ``#anchor`` and the absolute
    (``/...``) vs relative style. External and out-of-bundle links are left
    untouched, as are links whose rendered text would be unchanged.
    """
    bundle_root = bundle_root.resolve()
    doc_dir_old = doc_dir_old.resolve()
    doc_dir_new = doc_dir_new.resolve()
    dir_changed = doc_dir_old != doc_dir_new

    def _sub(m: re.Match[str]) -> str:
        target = m.group("target")
        anchor = m.group("anchor") or ""
        old_id = _resolve_target_id(target, doc_dir_old, bundle_root)
        if old_id is None:
            return m.group(0)
        new_id = id_map.get(old_id, old_id)
        if new_id == old_id and not dir_changed:
            return m.group(0)  # nothing to change; preserve verbatim
        new_path = concept_id_to_path(bundle_root, parse_concept_id(new_id))
        if target.startswith("/"):
            new_target = "/" + new_path.relative_to(bundle_root).as_posix()
        else:
            new_target = Path(os.path.relpath(new_path, doc_dir_new)).as_posix()
        if new_target == target:
            return m.group(0)
        return f"]({new_target}{anchor})"

    return _LINK_RE.sub(_sub, body)
```

```python
def _log_move(
    bundle_root: Path,
    from_path: Path,
    to_path: Path,
    frontmatter: dict,
    from_id: str,
    to_id: str,
) -> list[Path]:
    """Append Move entries: an arrival in the dest dir log, and (when the source
    directory differs) a departure in the source dir log. Returns the logs."""
    label = str(frontmatter.get("title") or to_path.stem)
    today = _today_iso()
    dest_log = to_path.parent / "log.md"
    _append_log(
        dest_log,
        f"* **Move**: Moved [{label}]({to_path.name}) from `{from_id}`.",
        today,
    )
    logs = [dest_log]
    if from_path.parent != to_path.parent:
        src_log = from_path.parent / "log.md"
        _append_log(
            src_log,
            f"* **Move**: Moved [{label}]({from_path.name}) to `{to_id}`.",
            today,
        )
        logs.append(src_log)
    return logs


def move_concept(
    bundle_root: Path, from_concept_id: str, to_concept_id: str
) -> dict:
    """Relocate a concept to a new id: rewrite links, reindex, and log.

    Moves the file from ``from_concept_id`` to ``to_concept_id``, rewriting the
    moved document's own relative links for its new directory. Inbound links from
    other concepts are rewritten separately (see ``_rewrite_inbound_links``).
    Preserves frontmatter (including ``timestamp``). Fails if the source is
    missing, the destination exists, or the two ids are equal.
    """
    from_path = _concept_path(bundle_root, from_concept_id)
    to_path = _concept_path(bundle_root, to_concept_id)
    if from_path == to_path:
        raise ValueError("source and destination are the same concept")
    if not from_path.exists():
        raise ValueError(f"concept not found: {from_concept_id}")
    if to_path.exists():
        raise ValueError(f"destination already exists: {to_concept_id}")

    from_id = "/".join(parse_concept_id(from_concept_id))
    to_id = "/".join(parse_concept_id(to_concept_id))
    id_map = {from_id: to_id}

    doc = OKFDocument.parse(from_path.read_text(encoding="utf-8"))
    new_body = rewrite_links(
        doc.body or "", from_path.parent, to_path.parent, bundle_root, id_map
    )
    to_path.parent.mkdir(parents=True, exist_ok=True)
    to_path.write_text(
        OKFDocument(frontmatter=doc.frontmatter, body=new_body).serialize(),
        encoding="utf-8",
    )
    from_path.unlink()

    relinked: list[str] = []  # inbound rewriting added in a later task

    logs = _log_move(bundle_root, from_path, to_path, doc.frontmatter, from_id, to_id)
    indexes = regenerate_indexes(bundle_root)
    return {
        "from": from_id,
        "to": to_id,
        "action": "moved",
        "relinked": relinked,
        "logs": [Path(p).relative_to(bundle_root).as_posix() for p in logs],
        "indexes": [Path(p).relative_to(bundle_root).as_posix() for p in indexes],
    }
```

```python
def _rewrite_inbound_links(
    bundle_root: Path, moved_id: str, id_map: dict[str, str]
) -> list[str]:
    """Rewrite links to the moved concept in every other concept document.

    Returns the ids of concepts whose body changed (sorted). The moved concept
    itself (now at ``moved_id``) is skipped -- its body was rewritten on write.
    """
    bundle = Bundle.load(bundle_root)
    relinked: list[str] = []
    for c in bundle.concepts:
        if c.id == moved_id:
            continue
        path = concept_id_to_path(bundle_root, parse_concept_id(c.id))
        new_body = rewrite_links(c.body, path.parent, path.parent, bundle_root, id_map)
        if new_body != c.body:
            path.write_text(
                OKFDocument(
                    frontmatter=c.document.frontmatter, body=new_body
                ).serialize(),
```
